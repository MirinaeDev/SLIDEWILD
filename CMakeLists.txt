cmake_minimum_required(VERSION 3.24)
project(Slidewild VERSION 0.35.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
if(NOT WIN32 AND NOT APPLE)
    message(FATAL_ERROR "Supported platforms: Windows x64 and macOS Apple Silicon.")
endif()
if(APPLE)
    if(NOT CMAKE_OSX_ARCHITECTURES STREQUAL "arm64")
        message(FATAL_ERROR "Configure with -DCMAKE_OSX_ARCHITECTURES=arm64 (Apple Silicon only).")
    endif()
elseif(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Windows x64 requires a 64-bit compiler.")
endif()
option(DH_FETCH_SDL "Build the bundled, pinned SDL3 source (no network required)" ${APPLE})
if(DH_FETCH_SDL)
    include(FetchContent)
    set(SDL_SHARED OFF CACHE BOOL "" FORCE)
    set(SDL_STATIC ON CACHE BOOL "" FORCE)
    set(SDL_TEST_LIBRARY OFF CACHE BOOL "" FORCE)
    set(SDL_TESTS OFF CACHE BOOL "" FORCE)
    set(SDL_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(SDL_INSTALL OFF CACHE BOOL "" FORCE)
    set(SDL_HIDAPI_LIBUSB OFF CACHE BOOL "" FORCE)
    FetchContent_Declare(SDL3
        URL "${CMAKE_SOURCE_DIR}/vendor/SDL3-3.4.18.tar.gz"
        URL_HASH SHA256=c4b08b950bd29d83caae0ab8d884298d27046a9d30700fb42841c50992665c90
        DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
    FetchContent_MakeAvailable(SDL3)
    set(DH_SDL SDL3::SDL3-static)
else()
    find_package(SDL3 3.4 CONFIG REQUIRED)
    set(DH_SDL SDL3::SDL3)
endif()
function(dh_target target)
    target_compile_options(${target} PRIVATE $<$<CXX_COMPILER_ID:GNU,Clang,AppleClang>:-Wall;-Wextra;-Wpedantic>)
    if(MINGW)
        target_link_options(${target} PRIVATE -static-libgcc -static-libstdc++)
    endif()
endfunction()
add_library(downhill_render STATIC src/renderer.cpp src/world.cpp)
target_include_directories(downhill_render PUBLIC src)
target_link_libraries(downhill_render PUBLIC ${DH_SDL})
dh_target(downhill_render)
if(APPLE)
    add_executable(downhill MACOSX_BUNDLE src/main.cpp)
    set_target_properties(downhill PROPERTIES
        OUTPUT_NAME Slidewild
        MACOSX_BUNDLE_INFO_PLIST "${CMAKE_SOURCE_DIR}/cmake/Info.plist.in"
        MACOSX_BUNDLE_BUNDLE_NAME "SLIDEWILD"
        MACOSX_BUNDLE_GUI_IDENTIFIER "org.damienmiri.slidewild"
        MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}"
        MACOSX_BUNDLE_BUNDLE_VERSION "35")
    set(DH_RESOURCES "$<TARGET_BUNDLE_CONTENT_DIR:downhill>/Resources")
    set(DH_SHADER_DIR "${CMAKE_SOURCE_DIR}/shaders/metal")
    set(DH_SHADER_EXT msl)
else()
    add_executable(downhill src/main.cpp)
    set_target_properties(downhill PROPERTIES OUTPUT_NAME Slidewild)
    set(DH_RESOURCES "$<TARGET_FILE_DIR:downhill>")
    set(DH_SHADER_DIR "${CMAKE_BINARY_DIR}/shaders")
    set(DH_SHADER_EXT dxil)
    find_program(DXC_EXECUTABLE dxc HINTS "${CMAKE_SOURCE_DIR}/.tools/dxc/bin/x64" REQUIRED)
    foreach(STAGE vs ps shadow)
        set(PROFILE ps_6_0)
        set(ENTRY "${STAGE}_main")
        if(STAGE STREQUAL "vs")
            set(PROFILE vs_6_0)
        elseif(STAGE STREQUAL "shadow")
            set(ENTRY shadow_ps)
        endif()
        add_custom_command(OUTPUT "${DH_SHADER_DIR}/world.${STAGE}.dxil"
            COMMAND ${CMAKE_COMMAND} -E make_directory "${DH_SHADER_DIR}"
            COMMAND "${DXC_EXECUTABLE}" -T ${PROFILE} -E ${ENTRY} -O3
                -Fo "${DH_SHADER_DIR}/world.${STAGE}.dxil" "${CMAKE_SOURCE_DIR}/shaders/world.hlsl"
            DEPENDS shaders/world.hlsl VERBATIM)
        list(APPEND SHADERS "${DH_SHADER_DIR}/world.${STAGE}.dxil")
    endforeach()
    add_custom_target(shaders ALL DEPENDS ${SHADERS})
    add_dependencies(downhill shaders)
    if(TARGET SDL3::SDL3-shared AND NOT DH_FETCH_SDL)
        add_custom_command(TARGET downhill POST_BUILD
            COMMAND ${CMAKE_COMMAND} -E copy_if_different "$<TARGET_FILE:SDL3::SDL3-shared>" "$<TARGET_FILE_DIR:downhill>")
    endif()
endif()
target_link_libraries(downhill PRIVATE downhill_render)
dh_target(downhill)
add_custom_target(check_metal ALL
    COMMAND ${CMAKE_COMMAND} "-DROOT=${CMAKE_SOURCE_DIR}" -P "${CMAKE_SOURCE_DIR}/cmake/VerifyMetal.cmake" VERBATIM)
if(APPLE)
    add_dependencies(downhill check_metal)
endif()
# Runs on every build, even when only an asset changes; copy_if_different is incremental.
add_custom_target(stage_runtime ALL
    COMMAND ${CMAKE_COMMAND} "-DROOT=${CMAKE_SOURCE_DIR}" "-DDEST=${DH_RESOURCES}"
        "-DSHADER_DIR=${DH_SHADER_DIR}" "-DSHADER_EXT=${DH_SHADER_EXT}"
        -P "${CMAKE_SOURCE_DIR}/cmake/StageResources.cmake"
    DEPENDS downhill VERBATIM)
include(CTest)
if(BUILD_TESTING)
    foreach(TEST camera_projection physics race scene course_scene platform presentation)
        add_executable(${TEST}_tests "tests/${TEST}_tests.cpp")
        target_include_directories(${TEST}_tests PRIVATE src)
        if("${TEST}" STREQUAL "scene" OR "${TEST}" STREQUAL "course_scene")
            target_link_libraries(${TEST}_tests PRIVATE downhill_render)
        endif()
        if("${TEST}" STREQUAL "platform")
            target_link_libraries(${TEST}_tests PRIVATE ${DH_SDL})
        endif()
        dh_target(${TEST}_tests)
        add_test(NAME ${TEST} COMMAND ${TEST}_tests)
        set_tests_properties(${TEST} PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}")
    endforeach()
    add_test(NAME metal_fingerprints COMMAND ${CMAKE_COMMAND} "-DROOT=${CMAKE_SOURCE_DIR}" -P "${CMAKE_SOURCE_DIR}/cmake/VerifyMetal.cmake")
endif()
